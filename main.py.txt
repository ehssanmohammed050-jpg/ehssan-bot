import threading

from flask import Flask

if not BOT_TOKEN:
    print("ERROR: BOT_TOKEN not set")
    exit(1)

DB_PATH = "users.db"

def init_db():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("CREATE TABLE IF NOT EXISTS users (chat_id INTEGER PRIMARY KEY)")
    conn.commit()
    conn.close()

def save_user(chat_id):
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("INSERT OR IGNORE INTO users (chat_id) VALUES (?)", (chat_id,))
    conn.commit()
    conn.close()

def get_users():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("SELECT chat_id FROM users")
    rows = c.fetchall()
    conn.close()
    return [r[0] for r in rows]

def send_message(chat_id, text):
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendMessage"
    try:
        requests.post(url, json={"chat_id": chat_id, "text": text})
    except Exception as e:
        print(f"Failed to send to {chat_id}: {e}")

def bot_loop():
    offset = 0
    print("Bot started...")
    while True:
        try:
            url = f"https://api.telegram.org/bot{BOT_TOKEN}/getUpdates?timeout=30&offset={offset}"
            resp = requests.get(url, timeout=35).json()
            if "result" in resp:
                for upd in resp["result"]:
                    offset = upd["update_id"] + 1
                    if "message" in upd:
                        chat_id = upd["message"]["chat"]["id"]
                        text = upd["message"].get("text", "")
                        save_user(chat_id)
                        if text in ["/start", "start"]:
                            send_message(chat_id, "أهلا يا إحسان! 🌸 انا بقيت شغال هنا في Render، ح أرسل ليك رسائل تحفيزية كل 3 ساعات من 8 الصباح لـ 10 المساء ان شاء الله 💪")
                        else:
                            send_message(chat_id, "تمام يا إحسان، رسالتك وصلتني ✅ انا ح أكون معاك طول اليوم برسائل تحفيزية!")
        except Exception as e:
            print(f"bot loop error: {e}")
            time.sleep(5)

def scheduler_loop():
    print("Scheduler started...")
    while True:
        try:
            now = datetime.now(KHARTOUM_TZ)
            # من 8 صباحا لـ 10 مساء
            if 8 <= now.hour <= 22:
                # كل 3 ساعات: 8, 11, 14, 17, 20
                if now.hour in [8, 11, 14, 17, 20] and now.minute == 0:
                    users = get_users()
                    msg = random.choice(MESSAGES)
                    for uid in users:
                        send_message(uid, msg)
                    # نوم دقيقة عشان ما يكرر
                    time.sleep(60)
            time.sleep(30)
        except Exception as e:
            print(f"scheduler error: {e}")
            time.sleep(30)

# Flask عشان Render يفتكر انو ويب سيرفس
app = Flask(__name__)

@app.route("/")
def home():
    return "Ehssan Bot is Running ✅"

if __name__ == "__main__":
    init_db()
    threading.Thread(target=bot_loop, daemon=True).start()
    threading.Thread(target=scheduler_loop, daemon=True).start()
    port = int(os.environ.get("PORT", 10000))
    app.run(host="0.0.0.0", port=port)
